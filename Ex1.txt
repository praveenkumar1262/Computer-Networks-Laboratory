#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define STARTFLAG  "01111110"
#define ADDRESS    "11111111"
#define CONTROL    "00000011"
#define FCS        "00000000000000000000000000000000"
#define ENDFLAG    "01111110"
#define ESC        "11100011"
#define FLAG_SIZE      9
#define ADDRESS_SIZE   9
#define CONTROL_SIZE   9
#define FCS_SIZE       33
struct Frame
{
    char startFlag[FLAG_SIZE];
    char address[ADDRESS_SIZE];
    char control[CONTROL_SIZE];
    char *data;
    int dataLength;
    char fcs[FCS_SIZE];
    char endFlag[FLAG_SIZE];
};
void initializeFrame(struct Frame *frame);
int calculateFrames(int totalBits, int frameSize);
int calculatePayload(int frameSize);
struct Frame bitStuffing(struct Frame frame);
struct Frame bitDestuffing(struct Frame frame);
struct Frame byteStuffing(struct Frame frame);
struct Frame byteDestuffing(struct Frame frame);
void stringToBinary(char input[], char output[]);
void decimalToBinary(int number, char binary[]);
void binaryToString(char input[], char output[]);
void readFile(char fileName[], char data[]);
void writeFile(char fileName[], char data[]);
void displayFrame(struct Frame frame);
void initializeFrame(struct Frame *frame)
{
    strcpy(frame->startFlag, STARTFLAG);
    strcpy(frame->address, ADDRESS);
    strcpy(frame->control, CONTROL);
    strcpy(frame->fcs, FCS);
    strcpy(frame->endFlag, ENDFLAG);
    frame->data = NULL;
    frame->dataLength = 0;
}
int calculatePayload(int frameSize)
{
    int payload;
    payload = frameSize - 64;
    if(payload <= 0)
    {
        printf("\nInvalid Frame Size!\n");
        printf("Frame size should be greater than 64 bits.\n");
        return 0;
    }
    return payload;
}
int calculateFrames(int totalBits, int frameSize)
{
    int payload;
    int frames;
    payload = calculatePayload(frameSize);
    if(payload == 0)
        return 0;
    frames = totalBits / payload;
    if(totalBits % payload != 0)
        frames++;
    return frames;
}
void displayFrame(struct Frame frame)
{
    printf("\n-------------------------------------------\n");
    printf("Start Flag : %s\n", frame.startFlag);
    printf("Address    : %s\n", frame.address);
    printf("Control    : %s\n", frame.control);
    printf("Data       : ");
    if(frame.data != NULL)
        printf("%s\n", frame.data);
    else
        printf("NULL\n");
    printf("Data Length: %d bits\n", frame.dataLength);
    printf("FCS        : %s\n", frame.fcs);
    printf("End Flag   : %s\n", frame.endFlag);
    printf("-------------------------------------------\n");
}
struct Frame bitStuffing(struct Frame frame)
{
    struct Frame stuffedFrame;
    int i, j;
    int oneCount = 0;
    strcpy(stuffedFrame.startFlag, frame.startFlag);
    strcpy(stuffedFrame.address, frame.address);
    strcpy(stuffedFrame.control, frame.control);
    strcpy(stuffedFrame.fcs, frame.fcs);
    strcpy(stuffedFrame.endFlag, frame.endFlag);
    stuffedFrame.data = (char *)malloc((frame.dataLength * 2) + 1);

    if(stuffedFrame.data == NULL)
    {
        printf("Memory Allocation Failed\n");
        exit(1);
    }

    i = 0;
    j = 0;

    while(frame.data[i] != '\0')
    {
        stuffedFrame.data[j] = frame.data[i];
        if(frame.data[i] == '1')
        {
            if(i == 0)
            {
                oneCount = 1;
            }
            else
            {
                if(frame.data[i - 1] == '0')
                    oneCount = 1;
                else
                    oneCount++;
            }
            if(oneCount == 5)
            {
                j++;
                stuffedFrame.data[j] = '0';
                oneCount = 0;
            }
        }
        else
        {
            oneCount = 0;
        }
        i++;
        j++;
    }

    stuffedFrame.data[j] = '\0';

    stuffedFrame.dataLength = strlen(stuffedFrame.data);
    stuffedFrame.data = (char *)realloc(stuffedFrame.data,
                           stuffedFrame.dataLength + 1);
    return stuffedFrame;
}

struct Frame byteStuffing(struct Frame frame)
{
    struct Frame stuffedFrame;
    int i, j;
    strcpy(stuffedFrame.startFlag, frame.startFlag);
    strcpy(stuffedFrame.address, frame.address);
    strcpy(stuffedFrame.control, frame.control);
    strcpy(stuffedFrame.fcs, frame.fcs);
    strcpy(stuffedFrame.endFlag, frame.endFlag);
    stuffedFrame.data = (char *)malloc((frame.dataLength * 2) + 1);
    if(stuffedFrame.data == NULL)
    {
        printf("Memory Allocation Failed\n");
        exit(1);
    }
    i = 0;
    j = 0;
    while(frame.data[i] != '\0')
    {
        if(strncmp(&frame.data[i], STARTFLAG, 8) == 0)
        {
            strcpy(&stuffedFrame.data[j], ESC);
            j += 8;

            strncpy(&stuffedFrame.data[j], STARTFLAG, 8);
            j += 8;

            i += 8;
        }
        else if(strncmp(&frame.data[i], ESC, 8) == 0)
        {
            strcpy(&stuffedFrame.data[j], ESC);
            j += 8;
            strncpy(&stuffedFrame.data[j], ESC, 8);
            j += 8;
            i += 8;
        }
        else
        {
            stuffedFrame.data[j] = frame.data[i];
            i++;
            j++;
        }
    }

    stuffedFrame.data[j] = '\0';

    stuffedFrame.dataLength = strlen(stuffedFrame.data);

    stuffedFrame.data = (char *)realloc(stuffedFrame.data,
                         stuffedFrame.dataLength + 1);

    return stuffedFrame;
}

struct Frame byteDestuffing(struct Frame frame)
{

    if(destuffedFrame.data == NULL)
    {
        printf("Memory Allocation Failed\n");
        exit(1);
    }
    i = 0;
    j = 0;
    while(frame.data[i] != '\0')
    {
        if(strncmp(&frame.data[i], ESC, 8) == 0 &&
           strncmp(&frame.data[i + 8], STARTFLAG, 8) == 0)
        {
            strncpy(&destuffedFrame.data[j], STARTFLAG, 8);

            j += 8;
            i += 16;
        }
        else if(strncmp(&frame.data[i], ESC, 8) == 0 &&
                strncmp(&frame.data[i + 8], ESC, 8) == 0)
        {
            strncpy(&destuffedFrame.data[j], ESC, 8);

            j += 8;
            i += 16;
        }

        else
        {
            destuffedFrame.data[j] = frame.data[i];
            i++;
            j++;
        }
    }
    destuffedFrame.data[j] = '\0';
    destuffedFrame.dataLength = strlen(destuffedFrame.data);
    destuffedFrame.data = (char *)realloc(destuffedFrame.data,
                           destuffedFrame.dataLength + 1);
    return destuffedFrame;
}
void readFile(char fileName[], char data[])
{
    FILE *fp;
    char ch;
    int i = 0;
    fp = fopen(fileName, "r");
    if(fp == NULL)
    {
        printf("\nUnable to Open File.\n");
        exit(1);
    }
    while((ch = fgetc(fp)) != EOF)
    {
        if(ch != '\n' && ch != ' ')
        {
            data[i] = ch;
            i++;
        }
    }
    data[i] = '\0';
    fclose(fp);
}
void writeFile(char fileName[], char data[])
{
    FILE *fp;

    fp = fopen(fileName, "w");

    if(fp == NULL)
    {
        printf("\nUnable to Create File.\n");
        exit(1);
    }

    fputs(data, fp);

    fclose(fp);

    printf("\nData Successfully Written to %s\n", fileName);
}
void stringToBinary(char input[], char output[]) {
    int i,j;
    output[0] = '\0';
    for (i = 0; input[i] != '\0'; i++) {
        for (j = 7; j >= 0; j--) {
            if ((input[i] >> j) & 1) {
                strcat(output, "1");
            } else {
                strcat(output, "0");
            }
        }
    }
}
void decimalToBinary(int number, char binary[]) {
    int index = 0;
    int i,j;
    if (number == 0) {
        strcpy(binary, "0");
        return;
    }
    char temp[33];
    while (number > 0) {
        temp[index++] = (number % 2) + '0';
        number /= 2;
    }
    int pos = 0;
    for (i = index - 1; i >= 0; i--) {
        binary[pos++] = temp[i];
    }
    binary[pos] = '\0';
}
void binaryToString(char input[], char output[]) {
    int len = strlen(input);
    int i,j;
    int outIndex = 0;
    for (i = 0; i < len; i += 8) {
        char ch = 0;
        for (j = 0; j < 8 && (i + j) < len; j++) {
            ch <<= 1;
            if (input[i + j] == '1') {
                ch |= 1;
            }
        }
        output[outIndex++] = ch;
    }
    output[outIndex] = '\0';
}
