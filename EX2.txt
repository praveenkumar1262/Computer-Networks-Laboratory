void twoparity_Sender(struct Data *f)
{
    int size = f->datalength;
    char input[2000];
    strcpy(input, f->data);
    int i = 0, j = 0;
    int rows = 0;
    int cols = 8;

    while(j < size)
    {
        i++;
        if(i == 7)
        {
            rows++;
            i = 0;
        }
        j++;
    }

    if(i != 0)
        rows++;

    int matrix[rows + 1][cols];
    int block[rows];

    for(i = 0; i <= rows; i++)
    {
        for(j = 0; j < cols; j++)
        {
            matrix[i][j] = 0;
        }
    }

    int s = 0;
    for(i = 0; i < rows; i++)
    {
        block[i] = 0;
        for(j = 0; j < cols - 1; j++)
        {
            if(s < size)
                matrix[i][j] = input[s] - '0';
            else
                matrix[i][j] = 0;

            s++;
        }
    }

    int arr[cols - 1];
    int index = 0;
    int count = 0;
    for(j = 0; j < cols - 1; j++)
    {
        count = 0;
        for(i = 0; i < rows; i++)
        {
            if(matrix[i][j] == 1)
                count++;
        }

        if(count % 2 == 1){
            arr[index] = 0;
        }
        else{
            arr[index] = 1;
        }
        index++;
    }

    for(i = 0; i < cols - 1; i++)
    {
        matrix[rows][i] = arr[i];
    }

    int arr1[rows];
    index = 0;
    for(i = 0; i < rows; i++)
    {
        count = 0;
        for(j = 0; j < cols - 1; j++)
        {
            if(matrix[i][j] == 1)
                count++;
        }


        if(count % 2 == 1){
            arr1[index] = 0;
        }
        else{
            arr1[index] = 1;
        }
        index++;
    }

    for(i = 0; i < rows; i++)
    {
        matrix[i][cols - 1] = arr1[i];
    }
    count = 0;

    for(i = 0; i < cols - 1; i++)
    {
        if(matrix[rows][i] == 1)
            count++;
    }

    if(count % 2 == 1){
        matrix[rows][cols - 1] = 0;
    }
    else{
        matrix[rows][cols - 1] = 1;
    }

        for(j = 0; j < cols; j++)
        {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");

        printf("File cannot be opened.\n");
        return;

    {
        for(j = 0; j < cols; j++)
        {
            fprintf(fp, "%d", matrix[i][j]);
        }
    }

    fclose(fp);
    printf("\nData sent successfully.\n");
    char choice;
    int pos;
    char bit;

    printf("\nChange bit (y/n): ");
    scanf(" %c", &choice);
    fp = fopen("transmission.txt", "r+");
    while(choice == 'y' || choice == 'Y')
    {
        printf("Enter bit position to change : ");
        scanf("%d", &pos);
        fseek(fp, pos - 1, SEEK_SET);
        bit = fgetc(fp);
        fseek(fp, -1, SEEK_CUR);
        if(bit == '0')
            fputc('1', fp);
        else
            fputc('0', fp);

        printf("To change another bit: ");
        scanf(" %c", &choice);
    }

    fclose(fp);
}

    {
        if(ch == '0' || ch == '1')

    }

    {
        for(j = 0; j < cols; j++)

            matrix[i][j] = fgetc(fp) - '0';

    {
        for(j = 0; j < cols; j++)
        {
            temp[i][j] = 0;
        }
    }

    for(i = 0; i < rows - 1; i++)
    {

            temp[i][j] = matrix[i][j];

    }

    int count;
    for(j = 0; j < cols - 1; j++)
    {

        for(i = 0; i < rows - 1; i++)

            if(temp[i][j] == 1)
                count++;

                if(count % 2 == 1){
            temp[rows - 1][j] = 0;
        }
        else{
            temp[rows - 1][j] = 1;
        }
    }

    for(i = 0; i < rows - 1; i++)
    {
        count = 0;
        for(j = 0; j < cols - 1; j++)
        {
            if(temp[i][j] == 1)
                count++;
        }
                if(count % 2 == 1){
            temp[i][cols - 1] = 0;
        }
        else{
            temp[i][cols - 1] = 1;
        }
    }

    count = 0;
    for(i = 0; i < cols - 1; i++)
    {
        if(temp[rows - 1][i] == 1)
            count++;
    }
        if(count % 2 == 1){
        temp[rows - 1][cols - 1] = 0;
    }
    else{
        temp[rows - 1][cols - 1] = 1;
    }

    int rowError = -1;
    int colError = -1;
    int rowCount = 0;
    int colCount = 0;

    for(i = 0; i < rows - 1; i++)
    {
        if(temp[i][cols - 1] != matrix[i][cols - 1])
        {
            rowError = i;
            rowCount++;
        }
    }

    for(j = 0; j < cols - 1; j++)
    {
        if(temp[rows - 1][j] != matrix[rows - 1][j])
        {
            colError = j;
            colCount++;
        }
    }

    if(temp[rows - 1][cols - 1] != matrix[rows - 1][cols - 1])
    {
        printf("\nOverall parity mismatch\n");
    }

    printf("\nRECALCULATED ODD PARITY MATRIX\n\n");

    {
        for(j = 0; j < cols; j++)
        {
            printf("%d ", temp[i][j]);
        }
        printf("\n");
    }

    if(rowCount == 0 && colCount == 0)
    {
        if(temp[rows - 1][cols - 1] != matrix[rows - 1][cols - 1])
        {
            printf("\nError in Overall Parity Bit\n");
        }
        else
        {
            printf("\nNo Error\n");
        }
    }
    else if(rowCount == 1 && colCount == 1)
    {
        printf("\nSingle Bit Error Detected\n");
        printf("Error at Row %d Column %d\n", rowError + 1, colError + 1);
    }
    else
    {
        printf("\nMultiple Bit Error Detected\n");
    }
}

#ifndef HEAD_H
#define HEAD_H

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_SIZE 1000
#define K 8

struct Frame {
   char data[MAX_SIZE];
   int datalength;
};

void readFile(char fileName[], char data[]);
void writeFile(char fileName[], char data[]);

/* Checksum */
void checksumSender(char data[]);
void checksumReceiver(void);
void readFile(char fileName[], char data[])
{
    FILE *fp;
    char ch;
    int i = 0;
    fp = fopen(fileName, "r");
    if(fp == NULL)
    {
        printf("\nUnable to Open File.\n");
        exit(1);
    }
    while((ch = fgetc(fp)) != EOF)
    {
        if(ch != '\n' && ch != ' ')
        {
            data[i++] = ch;
        }
    }
    data[i] = '\0';
    fclose(fp);
}

void writeFile(char fileName[], char data[])
{
    FILE *fp;
    fp = fopen(fileName,"w");
    if(fp == NULL)
    {
        printf("\nUnable to Create File.\n");
        exit(1);
    }
    fputs(data,fp);
    fclose(fp);
    printf("\nData Written Successfully to %s\n",fileName);
}

void checksumSender(char data[])
{
    FILE *fp;
    int i, j;
    int rows;
    int length;
    int block[100];
    int sum = 0;
    int carry;
    int checksum;
    length = strlen(data);
    rows = length / K;
    if(length % K != 0)
        rows++;
    printf("\nCHECKSUM SENDER\n");
    printf("\nData Blocks\n\n");

    for(i = 0; i < rows; i++)
    {
        block[i] = 0;
        for(j = 0; j < K; j++)
        {
            int pos = i * K + j;
            block[i] <<= 1;
            if(pos < length)
            {
                if(data[pos] == '1')
                    block[i] |= 1;
                printf("%c", data[pos]);
            }
            else
            {
                printf("0");
            }
        }
        printf("\n");
    }

    printf("\nBinary Addition\n");
    for(i = 0; i < rows; i++)
    {
        sum += block[i];
        while(sum > 255)
        {
            carry = sum >> 8;
            sum = (sum & 255) + carry;
        }
    }
    printf("\nSum : ");
    for(i = 7; i >= 0; i--)
    {
        printf("%d", (sum >> i) & 1);
    }

    checksum = (~sum) & 255;
    printf("\nChecksum : ");
    for(i = 7; i >= 0; i--)
    {
        printf("%d", (checksum >> i) & 1);
    }

    fp = fopen("transmission.txt", "w");
    if(fp == NULL)
    {
        printf("\nUnable to Create transmission.txt\n");
        exit(1);
    }
    fprintf(fp, "%s", data);
    for(i = 7; i >= 0; i--)
    {
        fprintf(fp, "%d", (checksum >> i) & 1);
    }
    fclose(fp);
    printf("\n\nTransmission File Created Successfully.\n");
}

void checksumReceiver(void)
{
    FILE *fp;
    char received[MAX_SIZE];
    char data[MAX_SIZE];
    char checksumBits[9];
    int length;
    int dataLength;
    int rows;
    int block[100];
    int receivedChecksum = 0;
    int i, j;
    int sum = 0;
    int carry;
    fp = fopen("transmission.txt","r");
    if(fp == NULL)
    {
        printf("\nUnable to Open transmission.txt\n");
        exit(1);
    }
    fscanf(fp,"%s",received);
    fclose(fp);
    length = strlen(received);

    dataLength = length - K;
    strncpy(data, received, dataLength);
    data[dataLength] = '\0';
    strcpy(checksumBits, received + dataLength);
    rows = dataLength / K;
    if(dataLength % K != 0)
        rows++;
    printf("\nCHECKSUM RECEIVER\n");
    printf("\nReceived Data Blocks\n\n");

    for(i = 0; i < rows; i++)
    {
        block[i] = 0;
        for(j = 0; j < K; j++)
        {
            int pos = i * K + j;
            block[i] <<= 1;
            if(pos < dataLength)
            {
                if(data[pos] == '1')
                    block[i] |= 1;
                printf("%c", data[pos]);
            }
            else
            {
                printf("0");
            }
        }
        printf("\n");
    }
    printf("\nReceived Checksum : %s\n", checksumBits);

    for(i = 0; i < K; i++)
    {
        receivedChecksum <<= 1;
        if(checksumBits[i] == '1')
            receivedChecksum |= 1;
    }

    for(i = 0; i < rows; i++)
    {
        sum += block[i];
        while(sum > 255)
        {
            carry = sum >> 8;
            sum = (sum & 255) + carry;
        }
    }

    sum += receivedChecksum;
    while(sum > 255)
    {
        carry = sum >> 8;
        sum = (sum & 255) + carry;
    }

    printf("\nFinal Sum : ");
    for(i = 7; i >= 0; i--)
    {
        printf("%d", (sum >> i) & 1);
    }

    sum = (~sum) & 255;
    printf("\nOne's Complement : ");
    for(i = 7; i >= 0; i--)
    {
        printf("%d", (sum >> i) & 1);
    }
    if(sum == 0)
    {
        printf("\n\nNo Error Detected.\n");
    }
    else
    {
        printf("\n\nError Detected.\n");
    }
}

#include<stdio.h>
#include<stdlib.h>
#include<string.h>
#define MAX 5000
#define FLAG "01111110"
struct Node
{
    int power;
    struct Node *next;
};
struct Frame
{
    char startFlag[9];
    char address[9];
    char control[9];
    char data[MAX];
    char fcs[MAX];
    char endFlag[9];
};
struct Node *createNode(int power);
void insertNode(struct Node **head,int power);
void deleteNode(struct Node **head,int power);
int searchNode(struct Node *head,int power);
void freeList(struct Node *head);
void binaryToPolynomial(char bits[],struct Node **head);
int highestPower(struct Node *head);
void polynomialDivision(struct Node **dividend,struct Node *divisor);
struct Node *createNode(int power)
{
    struct Node *newNode;
    newNode=(struct Node*)malloc(sizeof(struct Node));
    newNode->power=power;
    newNode->next=NULL;
    return newNode;
}
/* Insert node in descending order */
void insertNode(struct Node **head,int power)
{
    struct Node *newNode=createNode(power);
    if(*head==NULL)
    {
        *head=newNode;
        return;
    }
    if(power>(*head)->power)
    {
        newNode->next=*head;
        *head=newNode;
        return;
    }
    struct Node *temp=*head;
    while(temp->next!=NULL && temp->next->power>power)
    {
        temp=temp->next;
    }
    if(temp->power==power)
    {
        free(newNode);
        return;
    }
    if(temp->next!=NULL && temp->next->power==power)
    {
        free(newNode);
        return;
    }
    newNode->next=temp->next;
    temp->next=newNode;
}
void deleteNode(struct Node **head,int power)
{
    struct Node *temp=*head;
    struct Node *prev=NULL;
    while(temp!=NULL)
    {
        if(temp->power==power)
        {

            if(prev==NULL)
                *head=temp->next;
            else
                prev->next=temp->next;
            free(temp);
            return;
        }
        prev=temp;
        temp=temp->next;
    }
}
/*Search Power*/
int searchNode(struct Node *head,int power)
{
    while(head!=NULL)
    {
        if(head->power==power)
            return 1;
        head=head->next;
    }
    return 0;
}
void freeList(struct Node *head)
{
    struct Node *temp;
    while(head!=NULL)
    {
        temp=head;
        head=head->next;
        free(temp);
    }
}
void binaryToPolynomial(char bits[],struct Node **head){
    int len=strlen(bits);
    int i;
    for(i=0;i<len;i++){
        if(bits[i]=='1')
        {
            insertNode(head,len-i-1);
        }
    }
}
/* Highest power */
int highestPower(struct Node *head)
{
    if(head==NULL)
        return -1;
    return head->power;
}
void polynomialDivision(struct Node **dividend,struct Node *divisor)
{
    int divisorPower=highestPower(divisor);
    while(*dividend!=NULL &&highestPower(*dividend)>=divisorPower)
    {
        int difference;
        difference=highestPower(*dividend)-divisorPower;
        struct Node *temp=divisor;
        while(temp!=NULL)
        {
            int power;
            power=temp->power+difference;
            if(searchNode(*dividend,power))
            {
                deleteNode(dividend,power);
            }
            else
            {
                insertNode(dividend,power);
            }
            temp=temp->next;
        }
    }
}
void generateCRC(char data[],char generator[],char fcs[])
{
    int i;
    char padded[MAX];
    strcpy(padded,data);
    int gLength=strlen(generator);
    for(i=0;i<gLength-1;i++)
    {
        strcat(padded,"0");
    }
    struct Node *dividend=NULL;
    struct Node *divisor=NULL;
    binaryToPolynomial(padded,&dividend);
    binaryToPolynomial(generator,&divisor);
    polynomialDivision(&dividend,divisor);
    fcs[0]='\0';
    for(i=gLength-2;i>=0;i--)
    {
        if(searchNode(dividend,i))
            strcat(fcs,"1");
        else
            strcat(fcs,"0");
    }
    freeList(dividend);
    freeList(divisor);
}

void crcReceiver()
{
    FILE *fp;
    char received[MAX];
    char generator[MAX];
    printf("\nEnter Generator Polynomial for Receiver : ");
    scanf("%s",generator);
    fp=fopen("transmission.txt","r");
    if(fp==NULL)
    {
        printf("Transmission file not found\n");
        return;
    }
    fscanf(fp,"%s",received);
    fclose(fp);
    int length=strlen(received);
    int frameNo=1;
    int i=0;
    int framereceived;
    printf("\n\nRECEIVER\n");
    while(i<length)
    {

        char frame[MAX];
        char dataFCS[MAX];
        int start=i;
        /*
          Find starting flag
        */

        while(start<length-7)
        {

            if(strncmp(received+start,FLAG,8)==0)
            {
                break;
            }
            start++;
        }
        if(start>=length-7)
            break;
        int end=start+8;
        /*
          Find ending flag
        */

        while(end<length-7)
        {
            if(strncmp(received+end,FLAG,8)==0)
            {
                break;
            }
            end++;
        }
        if(end>=length-7)
            break;
        strncpy(frame,received+start+8,end-start-8);
        frame[end-start-8]='\0';
        strcpy(dataFCS,frame+16);
        int dataLength;
        dataLength=strlen(dataFCS);
        struct Node *dividend=NULL;
        struct Node *divisor=NULL;
        binaryToPolynomial(dataFCS,&dividend);
        binaryToPolynomial(generator,&divisor);
        polynomialDivision(&dividend,divisor);
        printf("\nFRAME %d\n",frameNo);
        printf("Data + FCS : %s\n",dataFCS);
        if(dividend==NULL)
        {
            printf("CRC CHECK : NO ERROR\n");
        }

        else
        {
            printf("CRC CHECK : ERROR DETECTED\n");
        }
        freeList(dividend);
        freeList(divisor);
        frameNo++;
        i=end+8;
    }
}
