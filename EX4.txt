	DIJKSTRA

#include <stdio.h>
#define INF 9999
#define MAX 20
void dijkstra(int graph[MAX][MAX], int n, int start) {
    int distance[MAX];
    int visited[MAX];
    int parent[MAX]; 
    int i, count, min, next;
    for(i = 0; i < n; i++) {
        distance[i] = INF;
        visited[i] = 0;
        parent[i] = -1; 
    }
    distance[start] = 0;
    parent[start] = start; 
    for(count = 0; count < n - 1; count++) {
        min = INF;
        next = -1;
        for(i = 0; i < n; i++) {
            if(!visited[i] && distance[i] < min) {
                min = distance[i];
                next = i;
            }
        }
        if(next == -1) break;
        visited[next] = 1;
        for(i = 0; i < n; i++) {
            if(!visited[i] && graph[next][i] != 0 && distance[next] != INF && distance[next] + graph[next][i] < distance[i]) {
                distance[i] = distance[next] + graph[next][i];
                parent[i] = next; 
            }
        }
    }
    printf("\nDijkstra's Algorithm Results");
    printf("\nVertex\tNextnode \tDistance\n");
    for(i = 0; i < n; i++) {
        if(distance[i] == INF) {
            printf("%d\tNone\t\tNo path\n", i);
        } else if (i == start) {
            printf("%d\t%d (Start)\t%d\n", i, start, distance[i]);
        } else {
            int curr = i;
            int path_nodes = 1;
            int first_reached = i;
            while(parent[curr] != start && parent[curr] != -1) {
                first_reached = parent[curr];
                curr = parent[curr];
                path_nodes++;
            }
            path_nodes++; 
            if(path_nodes == 2) {
                printf("%d\t%d\t\t%d\n", i, start, distance[i]);
            } else {
                printf("%d\t%d\t\t%d\n", i, first_reached, distance[i]);
            }
        }
    }
}
int main() {
    int graph[MAX][MAX] = {0};
    int n, i, j, weight, start;
    printf("Enter the number of vertices: ");
    if (scanf("%d", &n) != 1 || n <= 0 || n > MAX) {
        printf("Invalid number of vertices.\n");
        return 1;
    }
    printf("\nEnter the weight for each edge");
    printf("\n(Enter 0 if there is no edge)\n");
    for(i = 0; i < n; i++) {
        for(j = i + 1; j < n; j++) {
            printf("Weight between vertex %d and %d: ", i, j);
            scanf("%d", &weight);
            graph[i][j] = weight;
            graph[j][i] = weight;
        }
    }
    printf("\nEnter the starting vertex: ");
    scanf("%d", &start);
    if(start < 0 || start >= n) {
        printf("Invalid starting vertex.\n");
        return 1;
    }
    dijkstra(graph, n, start);
    return 0;
}

	BELLMANFORD

#include <stdio.h> 
#include <stdlib.h> 
#include <limits.h> 
struct Edge { 
    int src, dest, weight; 
}; 
struct Graph { 
    int V, E; 
    struct Edge* edge; 
}; 
struct Graph* createGraph(int V, int E) { 
    struct Graph* graph = (struct Graph*) malloc(sizeof(struct Graph)); 
    graph->V = V; 
    graph->E = 2 * E; 
    graph->edge = (struct Edge*) malloc(graph->E * sizeof(struct Edge)); 
    return graph; 
} 
void printPass(int dist[], int V, int passNum) { 
    int i; 
    if (passNum == 0) {
        printf("Initial Status: [");
    } else {
        printf("Pass %d Status: [", passNum);
    }
    for (i = 0; i < V; i++) { 
        if (dist[i] == INT_MAX) 
            printf(" INF"); 
        else 
            printf(" %d", dist[i]); 
        if (i < V - 1) 
            printf(","); 
    } 
    printf(" ]\n"); 
} 
void BellmanFord(struct Graph* graph, int src) {
    int i, j;
    int V = graph->V;
    int E = graph->E;
    int dist[V];
    int has_negative_cycle = 0;
    for (i = 0; i < V; i++)
        dist[i] = INT_MAX;
    dist[src] = 0;
    printf("\n--- Pass start to relax ---\n");
    printPass(dist, V, 0);
    for (i = 1; i < V; i++) {
        int updated = 0;
        for (j = 0; j < E; j++) {
            int u = graph->edge[j].src;
            int v = graph->edge[j].dest;
            int weight = graph->edge[j].weight;
            if (dist[u] != INT_MAX && dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                if (i == V) {
                    has_negative_cycle = 1;
                }
            }
        }
        printPass(dist, V, i);
    }
    if (has_negative_cycle) {
        printf("\nGraph contains a negative cycle!\n");
        return;
    }
    printf("\n--- Final Report ---\n");
    printf("Vertex Distance from Source\n");
    for (i = 0; i < V; ++i) {
        if (dist[i] == INT_MAX)
            printf("%d \t\t INF\n", i);
        else
            printf("%d \t\t %d\n", i, dist[i]);
    }
}
      
int main() { 
    int V, E, src, i; 
    printf("Enter number of vertices: "); 
    if (scanf("%d", &V) != 1) return 1;
    printf("Enter number of undirected edges: "); 
    if (scanf("%d", &E) != 1) return 1;
    struct Graph* graph = createGraph(V, E); 
    printf("Enter edges (source, destination, weight):\n"); 
    int edge_index = 0; 
    for (i = 0; i < E; i++) { 
        int u, v, w; 
        if (scanf("%d %d %d", &u, &v, &w) != 3) return 1;
        graph->edge[edge_index].src = u; 
        graph->edge[edge_index].dest = v; 
        graph->edge[edge_index].weight = w; 
        edge_index++; 
        graph->edge[edge_index].src = v; 
        graph->edge[edge_index].dest = u; 
        graph->edge[edge_index].weight = w; 
        edge_index++; 
    } 
    printf("Enter source vertex: "); 
    if (scanf("%d", &src) != 1) return 1;
    BellmanFord(graph, src); 
    free(graph->edge); 
    free(graph); 
    return 0; 
}
